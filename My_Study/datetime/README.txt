# 1. ОСНОВНЫЕ ИМПОРТЫ И СОЗДАНИЕ ОБЪЕКТОВ

# from datetime import datetime, date, time, timedelta
# datetime.now()             - текущие дата и время (локальные)
# datetime.utcnow()          - текущие дата и время по UTC
# date.today()               - только текущая дата (ГГГГ-ММ-ДД)
# datetime(2026, 10, 28, 19, 30, 0) - ручное создание даты и времени (год, месяц, день, час, мин, сек)
# date(2026, 10, 28)         - ручное создание только даты
# time(19, 30, 0)            - ручное создание только времени

# 2. ИЗВЛЕЧЕНИЕ КОМПОНЕНТОВ (СВОЙСТВА ОБЪЕКТА)

# dt = datetime.now()
# dt.year                    - год (например, 2026)
# dt.month                   - месяц (1-12)
# dt.day                     - день месяца (1-31)
# dt.hour                    - час (0-23)
# dt.minute                  - минута (0-59)
# dt.second                  - секунда (0-59)
# dt.microsecond             - микросекунда
# dt.weekday()               - день недели (0 = Понедельник, 6 = Воскресенье)
# dt.isoweekday()            - день недели (1 = Понедельник, 7 = Воскресенье)

# 3. ФОРМАТИРОВАНИЕ: ОБЪЕКТ В СТРОКУ (strftime)

# dt.strftime(format)        - преобразует datetime в строку по шаблону
# %Y - год (4 цифры)         [2026]
# %y - год (2 цифры)         [26]
# %m - месяц (2 цифры)       [01-12]
# %B - название месяца (полное)   [October]
# %b - название месяца (сокращ.)  [Oct]
# %d - день месяца (2 цифры) [01-31]
# %H - час в 24-ч формате    [00-23]
# %I - час в 12-ч формате    [01-12]
# %p - AM / PM
# %M - минуты (2 цифры)      [00-59]
# %S - секунды (2 цифры)     [00-59]
# %A - день недели (полный)  [Wednesday]
# %a - день недели (сокращ.) [Wed]
# Пример: dt.strftime("%Y-%m-%d %H:%M:%S") -> "2026-10-28 19:30:00"

# 4. ПАРСИНГ: СТРОКА В ОБЪЕКТ (strptime)

# datetime.strptime(string, format) - преобразует строку в объект datetime
# Пример: datetime.strptime("28/10/2026 19:30", "%d/%m/%Y %H:%M")

# 5. РАБОТА С ИНТЕРВАЛАМИ ВРЕМЕНИ (timedelta)

# timedelta(days=0, seconds=0, microseconds=0, milliseconds=0, minutes=0, hours=0, weeks=0)
# delta = timedelta(days=7, hours=3) - создание интервала
# future = dt + delta        - прибавить время к дате
# past = dt - delta          - вычесть время из даты
# diff = dt1 - dt2           - разница между двумя датами (возвращает timedelta)
# diff.days                  - разница в днях
# diff.total_seconds()       - полная разница в секундах

# 6. КОНВЕРТАЦИЯ В UNIX TIMESTAMP

# dt.timestamp()             - переводит datetime в секунды POSIX (float)
# datetime.fromtimestamp(ts) - преобразует Unix timestamp обратно в datetime

# 7. ЗАМЕНА КОМПОНЕНТОВ (replace)

# dt.replace(year=2030)      - создает новый объект с измененными полями без модификации исходного
