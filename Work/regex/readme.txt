# ==========================================
# ШПАРГАЛКА ПО РЕГУЛЯРНЫМ ВЫРАЖЕНИЯМ (REGEX)
# ==========================================

# 1. СПЕЦИАЛЬНЫЕ СИМВОЛЫ (КЛАССЫ)
# \d  - любая цифра [0-9]
# \D  - любая не-цифра
# \w  - любой буквенно-цифровой символ или подчеркивание [a-zA-Z0-9_] (плюс кириллица)
# \W  - любой не-словесный символ
# \s  - любой пробельный символ (пробел, табуляция \t, перевод строки \n)
# \S  - любой не-пробельный символ
# .   - абсолютно любой символ, кроме перевода строки (\n)

# 2. КВАНТИФИКАТОРЫ (СКОЛЬКО РАЗ ПОВТОРЯЕТСЯ)
# *   - 0 или более раз
# +   - 1 или более раз
# ?   - 0 или 1 раз (делает предыдущий элемент необязательным)
# {n} - ровно n раз
# {n,}  - n или более раз
# {,m}  - от 0 до m раз
# {n,m} - от n до m раз

# 3. ЯКОРИ (ГРАНИЦЫ СТРОКИ / СЛОВА)
# ^   - начало строки (или начала каждой строки при флаге re.M)
# $   - конец строки
# \b  - граница слова
# \B  - не граница слова

# 4. НАБОРЫ И АЛЬТЕРНАТИВА
# [abc]    - любой символ из указанных в скобках (a, b или c)
# [^abc]   - любой символ, КРОМЕ указанных в скобках (отрицание)
# [a-z]    - диапазон символов (от a до z)
# a|b      - или a, или b

# 5. ГРУППЫ И ЗАХВАТ
# (...)              - группа захвата (нумерованная)
# (?:...)            - группа без захвата (не сохраняется в памяти)
# (?P<name>...)      - именованная группа захвата (Python)

# 6. ФЛАГИ МОДУЛЯ re (Python)
# re.I (re.IGNORECASE) - игнорировать регистр (А == а)
# re.M (re.MULTILINE)  - многорежимный режим (^ и $ работают для каждой строки)
# re.S (re.DOTALL)     - точка (.) начинает захватывать и переводы строк (\n)

# 7. ОСНОВНЫЕ ФУНКЦИИ В PYTHON (модуль re)
# import re
# re.findall(pattern, string)          - возвращает список всех совпадений
# re.search(pattern, string)           - ищет первое совпадение (возвращает объект или None)
# re.match(pattern, string)            - ищет совпадение строго с НАЧАЛА строки
# re.sub(pattern, repl, string)        - заменяет паттерн на repl
# re.split(pattern, string)            - разбивает строку по паттерну
